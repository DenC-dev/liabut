% Phaistos Labs working-paper style for pandoc + XeLaTeX.
% Body in Gentium Book Plus, headings in Alegreya, the Phaistos paper-and-ink palette.
\usepackage{fontspec}
\usepackage{xcolor}
\definecolor{ink}{HTML}{2A2723}
\definecolor{ink2}{HTML}{57514A}
\definecolor{ink3}{HTML}{6B6458}
\definecolor{rulecol}{HTML}{C9BC9B}
\definecolor{paper}{HTML}{EDE5D0}
\definecolor{paperdeep}{HTML}{E3D8BD}
\definecolor{paperlight}{HTML}{F4EEDF}
\definecolor{pblue}{HTML}{1F6A9E}
\definecolor{ochre}{HTML}{B5451F}
\definecolor{ochretext}{HTML}{963A1C}

\setmainfont{GentiumBookPlus-Regular.ttf}[
  Path=fonts/, BoldFont=GentiumBookPlus-Bold.ttf, ItalicFont=GentiumBookPlus-Italic.ttf,
  BoldItalicFont=GentiumBookPlus-BoldItalic.ttf, Numbers=OldStyle, Ligatures=TeX]
\newfontfamily\heading{Alegreya-500.ttf}[
  Path=build/fonts/, ItalicFont=Alegreya-Italic-500.ttf, Numbers=Lining, Ligatures=TeX]
\newfontfamily\headingregular{Alegreya-400.ttf}[
  Path=build/fonts/, ItalicFont=Alegreya-Italic-400.ttf, Numbers=Lining, Ligatures=TeX]
\newfontfamily\tablefont{GentiumBookPlus-Regular.ttf}[
  Path=fonts/, BoldFont=GentiumBookPlus-Bold.ttf, ItalicFont=GentiumBookPlus-Italic.ttf, Numbers=Lining, Ligatures=TeX]

\color{ink}
\usepackage{microtype}
\frenchspacing
\setlength{\parindent}{1.1em}
\setlength{\parskip}{0pt}
\linespread{1.12}

% headings
\usepackage{titlesec}
\titleformat{\section}{\heading\Large\color{ink}\raggedright}{\color{ochretext}\thesection}{0.6em}{}
\titleformat{\subsection}{\heading\large\color{ink}\raggedright}{\color{ochretext}\thesubsection}{0.6em}{}
\titlespacing*{\section}{0pt}{1.6\baselineskip}{0.55\baselineskip}
\titlespacing*{\subsection}{0pt}{1.1\baselineskip}{0.4\baselineskip}

% captions
\usepackage{caption}
\DeclareCaptionFont{inkii}{\color{ink2}}
\captionsetup{font={small,inkii}, labelfont={color=ink,bf}, labelsep=period, justification=raggedright,
  singlelinecheck=false, skip=6pt}
\captionsetup[table]{position=top}

% tables
\usepackage{etoolbox}
\AtBeginEnvironment{tabular}{\small\tablefont\renewcommand{\arraystretch}{1.22}}
\AtBeginEnvironment{table}{\centering}
\usepackage{booktabs,array,calc}
\usepackage{colortbl}
\setlength{\heavyrulewidth}{0.8pt}
\setlength{\lightrulewidth}{0.4pt}
\arrayrulecolor{ink}

% lists
\usepackage{enumitem}
\setlist{itemsep=2pt, topsep=4pt}
\setlist[enumerate]{label={\color{ochretext}\arabic*.}}

% links
\AtBeginDocument{\hypersetup{colorlinks=true, linkcolor=pblue, urlcolor=pblue, citecolor=pblue}}

% running heads
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyfoot[L]{\headingregular\itshape\footnotesize\color{ink3}Liabut \quad·\quad Phaistos Labs working paper}
\fancyfoot[R]{\heading\footnotesize\color{ink2}\thepage}
\fancypagestyle{first}{\fancyhf{}%
  \fancyfoot[L]{\headingregular\itshape\footnotesize\color{ink3}Preprint, not peer reviewed. Data and code: github.com/DenC-dev/liabut}%
  \fancyfoot[R]{\heading\footnotesize\color{ink2}\thepage}}

% references: hanging indent
\newenvironment{refs}{\par\small\setlength{\parindent}{0pt}\setlength{\parskip}{3pt}%
  \everypar{\hangindent=1.4em\hangafter=1}\raggedright}{\par}

% figures stay near their text
\usepackage{float}
\floatplacement{figure}{htbp}
\renewcommand{\topfraction}{0.9}
\renewcommand{\textfraction}{0.08}
\renewcommand{\floatpagefraction}{0.75}
